const canvas = document.getElementById("gameCanvas"); const ctx = canvas.getContext("2d"); // Oyuncu 1 (Cody - Mavi) - WASD ile hareket eder const player1 = { x: 50, y: 400, size: 30, color: "#3498db", speed: 5 }; // Oyuncu 2 (May - Kırmızı) - Yön tuşlarıyla hareket eder const player2 = { x: 50, y: 200, size: 30, color: "#e74c3c", speed: 5 }; // Yardımlaşma Mekaniği Nesneleri const button = { x: 300, y: 420, width: 40, height: 10, pressed: false }; const gate = { x: 500, y: 150, width: 20, height: 100, open: false }; const finishLine = { x: 730, y: 200, width: 40, height: 100 }; const keys = {}; window.addEventListener("keydown", (e) => keys[e.key] = true); window.addEventListener("keyup", (e) => keys[e.key] = false); function update() { // Oyuncu 1 Hareket Kontrolü (WASD) if (keys["w"] || keys["W"]) player1.y -= player1.speed; if (keys["s"] || keys["S"]) player1.y += player1.speed; if (keys["a"] || keys["A"]) player1.x -= player1.speed; if (keys["d"] || keys["D"]) player1.x += player1.speed; // Oyuncu 2 Hareket Kontrolü (Yön Tuşları) if (keys["ArrowUp"]) player2.y -= player2.speed; if (keys["ArrowDown"]) player2.y += player2.speed; if (keys["ArrowLeft"]) player2.x -= player2.speed; if (keys["ArrowRight"]) player2.x += player2.speed; // Bulmaca Mekaniği: Oyuncu 1 butona basıyor mu? if (player1.x < button.x + button.width && player1.x + player1.size > button.x && player1.y < button.y + button.height && player1.y + player1.size > button.y) { button.pressed = true; gate.open = true; // Kapı açıldı! } else { button.pressed = false; gate.open = false; // Butondan çekilirse kapı kapanır } // Kapı Duvar Kontrolü (Kapı kapalıysa Oyuncu 2 geçemez) if (!gate.open) { if (player2.x + player2.size > gate.x && player2.x < gate.x + gate.width && player2.y + player2.size > gate.y && player2.y < gate.y + gate.height) { player2.x = gate.x - player2.size; // Kapıya çarpıp durur } } // Kazanma Durumu: İki oyuncu da bitiş çizgisine ulaştı mı? const p1InFinish = player1.x > finishLine.x; const p2InFinish = player2.x > finishLine.x; if (p1InFinish && p2InFinish) { alert("Tebrikler! Birlikte başardınız!"); player1.x = 50; player1.y = 400; player2.x = 50; player2.y = 200; } } function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); // Butonu Çiz ctx.fillStyle = button.pressed ? "#2ecc71" : "#f1c40f"; ctx.fillRect(button.x, button.y, button.width, button.height); // Kapıyı Çiz (Açıksa yeşil/ince, kapalıysa kırmızı/kalın duvar) if (!gate.open) { ctx.fillStyle = "#c0392b"; ctx.fillRect(gate.x, gate.y, gate.width, gate.height); } // Bitiş Çizgisini Çiz ctx.fillStyle = "#2ecc71"; ctx.fillRect(finishLine.x, finishLine.y, finishLine.width, finishLine.height); // Oyuncuları Çiz ctx.fillStyle = player1.color; ctx.fillRect(player1.x, player1.y, player1.size, player1.size); ctx.fillStyle = player2.color; ctx.fillRect(player2.x, player2.y, player2.size, player2.size); } function gameLoop() { update(); draw(); requestAnimationFrame(gameLoop); } gameLoop(); body { margin: 0; padding: 0; background-color: #1a1a1a; color: white; font-family: Arial, sans-serif; display: flex; justify-content: center; align-items: center; height: 100vh; } #game-container { text-align: center; } #info { margin-bottom: 15px; font-size: 18px; font-weight: bold; } canvas { border: 4px solid #fff; background-color: #2c3e50; box-shadow: 0 0 20px rgba(0,0,0,0.5); } Co-Op Yardımlaşma Oyunu (It Takes Two Esintili)
